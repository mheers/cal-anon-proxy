Fixture parsing pattern (from GO_TESTING.md §2):
```go
func loadFixture(t *testing.T, path string) *caldav.CalendarObject {
    t.Helper()
    f, err := os.Open(path)
    if err != nil { t.Fatalf("open fixture: %v", err) }
    defer f.Close()
    cal, err := ical.NewDecoder(f).Decode()
    if err != nil { t.Fatalf("decode fixture: %v", err) }
    return &caldav.CalendarObject{Data: cal}
}
```
